3. Kalkulator sederhana dengan switch (+, -, *, /) dan penanganan pembagian dengan nol. 
START

let nilai1, nilai2, operator, hasil;
nilai1 = 10;
nilai2 = 500;
operator = "+";

switch (operator) {

    case "+":
        hasil = nilai1 + nilai2;
        console.log("Hasil: " + hasil);
        break;

    case "-":
        hasil = nilai1 - nilai2;
        console.log("Hasil: " + hasil);
        break;

    case "*":
        hasil = nilai1 * nilai2;
        console.log("Hasil: " + hasil);
        break;

    case "/":
        if (nilai2 === 0) {
            console.log("Error: Tidak bisa dibagi nol");
        } else {
            hasil = nilai1 / nilai2;
            console.log("Hasil: " + hasil);
        }
        break;

    default:
        console.log("Operator tidak valid");
}

END
4. Penentu kategori BMI (berat / (tinggi * tinggi)): kurang (di bawah 18,5), normal 
(18,5-24,9), lebih (25-29,9), obesitas (30 ke atas). 

let berat, tinggi, bmi, keteragan;
berat = 170; // dalam kilogram
tinggi = 1.75; // dalam meter

bmi = berat / (tinggi * tinggi);

if (bmi < 18.5) {
    keteragan = "Kurang";
} else if (bmi < 25) {
    keteragan = "Normal";
} else if (bmi < 30) {
    keteragan = "Lebih";
} else {
    keteragan = "Obesitas";
}

console.log("Keteragan: " + keteragan);
